## Symmetry
$$
\begin{aligned}
(1)\quad & \sinh(-x)=-\sinh(x)\\[12pt]
(2)\quad & \cosh(-x)=\cosh(x)
\end{aligned}
$$

## Exponential
$$
\begin{aligned}
(1)\quad & \cosh(x)+\sinh(x)=e^{x}\\[12pt]
(2)\quad & \cosh(x)-\sinh(x)=e^{-x}\\[12pt]
(3)\quad & \dfrac{1+\tanh(x)}{1-\tanh(x)}=e^{2x}
\end{aligned}
$$

## Pythagorean-type identities
$$
\begin{aligned}
(1)\quad & \cosh^{2}(x)-\sinh^{2}(x)=1\\[12pt]
(2)\quad & 1-\tanh^{2}(x)=\operatorname{sech}^{2}(x)\\[12pt]
(3)\quad & 1-\operatorname{coth}^{2}(x)=-\operatorname{csch}^{2}(x)
\end{aligned}
$$

## Addition/Subtraction formulas
$$
\begin{aligned}
(1)\quad & \sinh(x\pm y)=\sinh(x)\cosh(y)\pm \cosh(x)\sinh(y)\\[12pt]
(2)\quad & \cosh(x\pm y)=\cosh(x)\cosh(y)\pm \sinh(x)\sinh(y)\\[12pt]
(3)\quad & \tanh(x\pm y)=\dfrac{\tanh(x)\pm \tanh(y)}{1\pm \tanh(x)\tanh(y)}
\end{aligned}
$$

## Double-angle identities
$$
\begin{aligned}
(1)\quad & \sinh(2x)=2\sinh(x)\cosh(x)\\[12pt]
(2)\quad & \cosh(2x)=\cosh^{2}(x)+\sinh^{2}(x)\\[12pt]
& \qquad\qquad\quad =2\cosh^{2}(x)-1\\[12pt]
& \qquad\qquad\quad =2\sinh^{2}(x)+1\\[12pt]
& \qquad\qquad\quad =\dfrac{1+\tanh^{2}(x)}{1-\tanh^{2}(x)}
\end{aligned}
$$

## Half-angle identities
$$
\begin{aligned}
(1)\quad & \sinh^{2}\!\left(\dfrac{x}{2}\right)=\dfrac{\cosh(x)-1}{2}\\[12pt]
(2)\quad & \cosh^{2}\!\left(\dfrac{x}{2}\right)=\dfrac{\cosh(x)+1}{2}
\end{aligned}
$$

## Miscellaneous
$$
\begin{aligned}
(1)\quad & \sinh(x)=\dfrac{2\tanh\!\left(\dfrac{x}{2}\right)}{1-\tanh^{2}\!\left(\dfrac{x}{2}\right)}\\[12pt]
& \qquad\qquad\quad =\dfrac{\tanh(x)}{\sqrt{1-\tanh^{2}(x)}}\\[12pt]
(2)\quad & \tanh(\ln(x))=\dfrac{x^{2}-1}{x^{2}+1}
\end{aligned}
$$

## Product-to-sum identities
$$
\begin{aligned}
(1)\quad & \sinh(x)\cosh(y)=\dfrac{1}{2}\big(\sinh(x-y)+\sinh(x+y)\big)\\[12pt]
(2)\quad & \cosh(x)\cosh(y)=\dfrac{1}{2}\big(\cosh(x-y)+\cosh(x+y)\big)\\[12pt]
(3)\quad & \sinh(x)\sinh(y)=\dfrac{1}{2}\big(\cosh(x+y)-\cosh(x-y)\big)
\end{aligned}
$$

## Sum-to-product identities
$$
\begin{aligned}
(1)\quad & \sinh(x)+\sinh(y)=2\sinh\!\left(\dfrac{x+y}{2}\right)\cosh\!\left(\dfrac{x-y}{2}\right)\\[12pt]
(2)\quad & \cosh(x)+\cosh(y)=2\cosh\!\left(\dfrac{x+y}{2}\right)\cosh\!\left(\dfrac{x-y}{2}\right)\\[12pt]
(3)\quad & \cosh(x)-\cosh(y)=2\sinh\!\left(\dfrac{x+y}{2}\right)\sinh\!\left(\dfrac{x-y}{2}\right)
\end{aligned}
$$
