$$
\begin{aligned}
\sin x &= \cos\!\left(\frac{\pi}{2}-x\right), &
\cos x &= \sin\!\left(\frac{\pi}{2}-x\right), &
\tan x &= \cot\!\left(\frac{\pi}{2}-x\right), \\[6pt]
\csc x &= \sec\!\left(\frac{\pi}{2}-x\right), &
\sec x &= \csc\!\left(\frac{\pi}{2}-x\right), &
\cot x &= \tan\!\left(\frac{\pi}{2}-x\right).
\end{aligned}
$$
