$$
\begin{aligned}
& \quad \text{(1)}\quad |(a-b)+b| \le |a-b|+|b| \\
& \qquad \quad \, \, |a-(b-b)| \le |a-b|+|b| \\
& \qquad \quad \, \, |a| \le |a-b|+|b| \\[4pt]
& \qquad \quad \, \, \Rightarrow\ \text{(2)}\quad |a|-|b| \le |a-b| \\[10pt]

& \quad \text{(3)}\quad |a+(b-a)| \le |a|+|b-a| \\
& \qquad \quad \, \, |(a-a)+b| \le |a|+|a-b| \\
& \qquad \quad \, \, |b| \le |a|+|a-b| \\[4pt]
& \qquad \quad \, \, \Rightarrow\ \text{(4)}\quad |a-b| \ge -\bigl(|a|-|b|\bigr) \\[10pt]

& \quad \text{Combining (2) and (4):} \\[2pt]
& \quad \text{(5)}\quad -|a-b| \le |a|-|b| \le |a-b| \\[8pt]

& \qquad \quad \, \, \Rightarrow\ \text{(6)}\quad \bigl||a|-|b|\bigr| \le |a-b| \, .
\end{aligned}
$$
