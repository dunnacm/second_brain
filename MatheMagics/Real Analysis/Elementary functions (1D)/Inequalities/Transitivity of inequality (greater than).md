$$
\begin{aligned}
\text{Given:}\quad &(1)\ x>y,\qquad (2)\ y>z \\[4pt]
\text{From given eqs.:}\quad 
&(3)\ x-y>0 \;\boldsymbol{\Leftrightarrow}\; (x-y)\in\mathbb{R}^+ \\
&(4)\ y-z>0 \;\boldsymbol{\Leftrightarrow}\; (y-z)\in\mathbb{R}^+ \\[6pt]
\text{From (3),(4):}\quad 
&(5)\ (x-y)+(y-z)\in\mathbb{R}^+ \;\boldsymbol{\Leftrightarrow}\; (x-y)+(y-z)>0 \;\boldsymbol{\Leftrightarrow}\; x-z>0
\end{aligned}
$$
