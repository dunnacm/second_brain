$$
\begin{aligned}
&\text{Given: } \quad (1)\ x>y,\quad (2)\ w\in\mathbb{R} \\[4pt]
&\text{From given eq. : } \quad (3)\ x-y>0 \\
& \qquad \qquad \qquad \qquad \quad \ \ \boldsymbol{\Leftrightarrow}\ x-y+w-w>0 \\
& \qquad \qquad \qquad \qquad \quad \ \ \boldsymbol{\Leftrightarrow}\ (x+w)-(y+w)>0 \\
& \qquad \qquad \qquad \qquad \quad \ \ \boldsymbol{\Leftrightarrow}\ x+w>y+w
\end{aligned}
$$
